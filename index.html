// piece/King.java • ROK (CASTLING) & ŞAH GÜVENLİĞİ DOĞRULAMA MOTORU
// =========================================================================
public boolean canMove(int targetCol, int targetRow) {
    if (isWithinBoard(targetCol, targetRow)) {
        // 1. Standart 1 Kare Şah Hareketi (N-S-E-W ve 4 Çapraz Yön)
        if (Math.abs(targetCol - preCol) + Math.abs(targetRow - preRow) == 1 ||
            Math.abs(targetCol - preCol) * Math.abs(targetRow - preRow) == 1) {
            if (isValidSquare(targetCol, targetRow)) return true;
        }

        // 2. ROK KURALI: Şah daha önce hiç oynamamış olmalıdır
        if (moved == false) {
            
            // --- ŞAH KANADI KISA ROK (O-O) ---
            if (targetCol == preCol + 2 && targetRow == preRow &&
                pieceIsOnStraightLine(targetCol, targetRow) == false &&
                GamePanel.isSquareAttacked(preCol, preRow, 1 - color) == false &&     // Şah aktif tehdit altında olamaz
                GamePanel.isSquareAttacked(preCol + 1, preRow, 1 - color) == false) { // Şahın geçtiği kare tehdit altında olamaz
                
                // Hedef karenin boş olduğunu doğrula
                boolean targetEmpty = true;
                for (Piece piece : GamePanel.simPieces) {
                    if (piece != this && piece.col == targetCol && piece.row == targetRow) {
                        targetEmpty = false; break;
                    }
                }
                if (!targetEmpty) return false;

                // 3 kare sağdaki Kale'nin (Rook) oynamamış olduğunu doğrula
                for (Piece piece : GamePanel.simPieces) {
                    if (piece.col == preCol + 3 && piece.row == preRow && piece.moved == false) {
                        GamePanel.castlingP = piece; // Kale taşını otomatik bağla
                        return true;
                    }
                }
            }

            // --- VEZİR KANADI UZUN ROK (O-O-O) ---
            if (targetCol == preCol - 2 && targetRow == preRow &&
                pieceIsOnStraightLine(targetCol, targetRow) == false &&
                GamePanel.isSquareAttacked(preCol, preRow, 1 - color) == false &&
                GamePanel.isSquareAttacked(preCol - 1, preRow, 1 - color) == false) {
                
                Piece knight = null, rook = null;
                for (Piece piece : GamePanel.simPieces) {
                    if (piece.col == preCol - 3 && piece.row == targetRow) knight = piece;
                    if (piece.col == preCol - 4 && piece.row == targetRow) rook = piece;
                }
                if (knight == null && rook != null && rook.moved == false) {
                    GamePanel.castlingP = rook;
                    return true;
                }
            }
        }
    }
    return false;
}

// piece/Pawn.java • GEÇERKEN ALMA (EN PASSANT), 2 KARE SÜRÜŞ & ÇAPRAZ ALMA
// =========================================================================
public boolean canMove(int targetCol, int targetRow) {
    if (isWithinBoard(targetCol, targetRow) && isSameSquare(targetCol, targetRow) == false) {
        
        // Renge göre dikey yön değeri: Beyaz yukarı (-1), Siyah aşağı (+1)
        int moveValue = (color == GamePanel.WHITE) ? -1 : 1;
        hittingP = getHittingP(targetCol, targetRow);

        // 1. Düz 1 Kare İleri Sürüş (Önü boş olmalı)
        if (targetCol == preCol && targetRow == preRow + moveValue && hittingP == null) {
            return true;
        }

        // 2. İlk Hamlede 2 Kare Sürüş (Araya taş girmemiş ve hiç oynamamış olmalı)
        if (targetCol == preCol && targetRow == preRow + moveValue * 2 &&
            hittingP == null && moved == false &&
            pieceIsOnStraightLine(targetCol, targetRow) == false) {
            return true;
        }

        // 3. Standart Çapraz Taş Alma (Diyagonalde rakip taş olmalı)
        if (Math.abs(targetCol - preCol) == 1 &&
            targetRow == preRow + moveValue &&
            hittingP != null && hittingP.color != color) {
            return true;
        }

        // 4. --- EN PASSANT (GEÇERKEN ALMA) ---
        // Yan karedeki rakip piyon bir önceki hamlede 2 kare zıplamışsa (twoStepped == true)
        // piyon onun arkasındaki boş kareye geçerek onu ele geçirebilir
        if (Math.abs(targetCol - preCol) == 1 && targetRow == preRow + moveValue) {
            for (Piece piece : GamePanel.simPieces) {
                if (piece.col == targetCol && piece.row == preRow && piece.twoStepped == true) {
                    hittingP = piece; // Hedefin arkasında kalan piyonu vurulacak taş olarak ata
                    return true;
                }
            }
        }
    }
    return false;
}

// piece/Knight.java • ATIN ENGELLERİ AŞMA & 'L' HAREKETİ GEOMETRİSİ
// =========================================================================
public boolean canMove(int targetCol, int targetRow) {
    if (isWithinBoard(targetCol, targetRow)) {
        
        // Atın hareketi 1:2 veya 2:1 oranında 'L' şeklindedir.
        // Sütun ve satır mutlak farklarının çarpımı tam olarak 2 olduğunda 'L' hareketi doğrulanır:
        // (ΔCol = 1 ve ΔRow = 2 -> 1 * 2 = 2) veya (ΔCol = 2 ve ΔRow = 1 -> 2 * 1 = 2)
        if (Math.abs(targetCol - preCol) * Math.abs(targetRow - preRow) == 2) {
            if (isValidSquare(targetCol, targetRow)) {
                // At diğer taşların üzerinden atlayabildiği için aradaki kareler taranmaz!
                return true;
            }
        }
    }
    return false;
}

// main/ChessLogic.java • 3 KADEMELİ ŞAH MAT & PAT (STALEMATE) MATRİSİ
// =========================================================================
public boolean isCheckmate() {
    Piece king = getKing(true);
    
    // 1. KADEME (Escape): Şah komşu güvenli bir kareye kaçabilir mi?
    if (kingCanMove(king)) return false;

    // 2. KADEME (Capture): Şahı doğrudan tehdit eden taş başka bir taşla alınabilir mi?
    for (Piece piece : gp.simPieces) {
        if (piece != king && piece.color != gp.currentColor &&
            piece.canMove(gp.checkingP.col, gp.checkingP.row)) {
            return false;
        }
    }

    // 3. KADEME (Block): Tehdit hattı (ışın) araya başka bir taş girilerek bloke edilebilir mi?
    int colDiff = Math.abs(gp.checkingP.col - king.col);
    int rowDiff = Math.abs(gp.checkingP.row - king.row);

    // Dikey saldırı hattı kontrolü
    if (colDiff == 0) {
        int start = Math.min(gp.checkingP.row, king.row) + 1;
        int end = Math.max(gp.checkingP.row, king.row);
        for (int r = start; r < end; r++) {
            for (Piece piece : gp.simPieces) {
                if (piece != king && piece.color != gp.currentColor &&
                    piece.canMove(gp.checkingP.col, r)) return false;
            }
        }
    }
    // Yatay ve Çapraz ışın blokaj algoritmaları...

    // 3 savunma kademesi de başarısızsa -> ŞAH MAT (CHECKMATE)!
    return true;
}

public void applyReplayMoves(int targetCount) {
    resetReplayBoard();
    animLastNotation = null;

    // Hedeflenen adıma kadar tüm hamleleri sanal tahta üzerinde işlet
    for (int i = 0; i < targetCount && i < currentReplayMoves.size(); i++) {
        String notation = currentReplayMoves.get(i);
        animLastNotation = notation;
        if (notation.equals("O-O") || notation.equals("O-O-O")) {
            applyCastlingReplay(notation, i, replayPieces);
        } else {
            parseAndApplyMove(notation, replayPieces);
        }
    }

    // Taşların pixel koordinatlarını tahta pozisyonuna senkronize et
    for (Piece p : replayPieces) {
        p.x = p.getX(p.col);
        p.y = p.getY(p.row);
    }
    replayMoveIndex = targetCount;
}

// main/DatabaseManager.java • SQLITE İLİŞKİSEL 'chess.db' KAYIT MOTORU

public static void saveGame(String result, ArrayList<String> moveLog) {
    if (conn == null) return;
    try {
        // 1. 'games' Tablosuna Genel Maç Bilgisini Ekle ve Otomatik ID'yi Al
        PreparedStatement gameStmt = conn.prepareStatement(
                "INSERT INTO games (result) VALUES (?)", Statement.RETURN_GENERATED_KEYS);
        gameStmt.setString(1, result);
        gameStmt.executeUpdate();

        ResultSet rs = gameStmt.getGeneratedKeys();
        int gameId = rs.next() ? rs.getInt(1) : -1;
        rs.close();
        gameStmt.close();

        // 2. 'moves' Tablosuna Her Bir Hamleyi İlişkisel Olarak (FOREIGN KEY) Kaydet
        PreparedStatement moveStmt = conn.prepareStatement(
                "INSERT INTO moves (game_id, move_number, half, notation) VALUES (?, ?, ?, ?)");

        for (int i = 0; i < moveLog.size(); i++) {
            moveStmt.setInt(1, gameId);
            moveStmt.setInt(2, (i / 2) + 1); // Hamle Numarası (1, 2, 3...)
            moveStmt.setInt(3, i % 2);       // 0: Beyaz, 1: Siyah
            moveStmt.setString(4, moveLog.get(i)); // Standart Notasyon (e4, Nf3, O-O...)
            moveStmt.executeUpdate();
        }
        moveStmt.close();
    } catch (SQLException e) { e.printStackTrace(); }
}
